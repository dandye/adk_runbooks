# Implementation Plan: Converting Runbooks to Skills with Progressive Disclosure

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Transform monolithic prompt-concatenated runbooks into standard modular Agent Skills (`SKILL.md`) with a lightweight 3-tier progressive disclosure engine (`skills/registry.py`, `load_skill` tool) across the entire ADK multi-agent SOC system.

**Architecture:** Centralized Skill Registry (`skills/registry.py`) indexes categorized `SKILL.md` packages with SDO YAML frontmatter. Agent system prompts are dynamically injected with a compact Available Skills catalog (`Use when...`), and agents retrieve full instructions on demand via the `load_skill` tool.

**Tech Stack:** Python 3.13, Google ADK (`google.adk`), PyYAML, pytest.

**Spec:** `docs/superpowers/specs/2026-08-20-convert-runbooks-to-skills-progressive-disclosure-design.md`

## Global Constraints
- Every skill must have a valid `SKILL.md` with YAML frontmatter containing `name`, `description` (starting with `"Use when..."`), and `category`.
- System prompts must NOT concatenate entire runbooks. They must only include the compact catalog generated by `SkillRegistry.get_skill_catalog()`.
- All sub-agents and the manager must receive `load_skill` and `list_available_skills` tools.
- All existing benchmark tests in `evals/tests/` must pass cleanly.

```python
# skills/registry.py
from dataclasses import dataclass
from pathlib import Path
import re
import yaml

@dataclass
class SkillMetadata:
    name: str
    description: str
    category: str
    path: Path
    version: str = "1.0.0"

class SkillRegistry:
    def __init__(self, skills_dir: Path | str | None = None):
        if skills_dir is None:
            self.skills_dir = Path(__file__).resolve().parent
        else:
            self.skills_dir = Path(skills_dir).resolve()
        self.skills: dict[str, SkillMetadata] = {}
        self.scan_skills()

    def scan_skills(self) -> dict[str, SkillMetadata]:
        self.skills.clear()
        if not self.skills_dir.exists():
            return self.skills

        for skill_path in self.skills_dir.rglob("SKILL.md"):
            try:
                content = skill_path.read_text(encoding="utf-8")
                metadata = self._parse_frontmatter(content, skill_path)
                if metadata:
                    self.skills[metadata.name] = metadata
                    # Also register alternative naming (e.g. triage_alerts vs triage-alerts)
                    alt_name = metadata.name.replace("-", "_")
                    if alt_name != metadata.name:
                        self.skills[alt_name] = metadata
            except Exception as e:
                print(f"Warning: Failed to parse skill at {skill_path}: {e}")
        return self.skills

    def _parse_frontmatter(self, content: str, path: Path) -> SkillMetadata | None:
        match = re.match(r"^---\s*\n(.*?)\n---\s*\n(.*)$", content, re.DOTALL)
        if not match:
            # Fallback for name from parent folder
            folder_name = path.parent.name
            return SkillMetadata(
                name=folder_name,
                description=f"Skill for {folder_name}",
                category=path.parent.parent.name,
                path=path,
            )
        fm_text = match.group(1)
        fm_data = yaml.safe_load(fm_text) or {}
        name = fm_data.get("name", path.parent.name)
        description = fm_data.get("description", "").strip()
        category = fm_data.get("category", path.parent.parent.name)
        version = str(fm_data.get("version", "1.0.0"))
        return SkillMetadata(
            name=name,
            description=description,
            category=category,
            path=path,
            version=version,
        )

    def get_skill(self, name: str) -> SkillMetadata | None:
        norm_name = name.strip()
        if norm_name in self.skills:
            return self.skills[norm_name]
        alt = norm_name.replace("_", "-")
        if alt in self.skills:
            return self.skills[alt]
        return None

    def get_skill_content(self, name: str) -> str:
        meta = self.get_skill(name)
        if not meta or not meta.path.exists():
            return f"Error: Skill '{name}' not found in registry."
        return meta.path.read_text(encoding="utf-8")

    def get_skill_catalog(self, skill_names: list[str] | None = None) -> str:
        lines = [
            "### Available Skills (Progressive Disclosure)",
            "You have access to the following skills. When assigned a matching task or when a trigger condition is met, call the `load_skill(skill_name)` tool to retrieve complete step-by-step procedures before execution:\n",
        ]
        items = []
        if skill_names:
            seen = set()
            for s in skill_names:
                meta = self.get_skill(s)
                if meta and meta.name not in seen:
                    items.append(meta)
                    seen.add(meta.name)
        else:
            seen = set()
            for meta in self.skills.values():
                if meta.name not in seen:
                    items.append(meta)
                    seen.add(meta.name)

        if not items:
            return ""

        for item in sorted(items, key=lambda x: (x.category, x.name)):
            lines.append(f"- **`{item.name}`**: {item.description}")

        return "\n".join(lines)
```
